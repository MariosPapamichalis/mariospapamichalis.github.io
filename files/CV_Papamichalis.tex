\begin{rSection}{Publications: Accepted, In Press, and Published}

\pubtitle{Educational Intervention Can Rewire Village Social Networks} \hfill {\em 2026}\\
\textit{(with Laura Forastiere, Edoardo M.\ Airoldi, and Nicholas A.\ Christakis)}\\
{\small \textbf{Accepted, \emph{Nature}.}}\\
{\small \textbf{Abstract.} Using a two-stage randomized public-health intervention in 110 remote Honduran villages (8,331 individuals), we quantify how an informational intervention alters pre-existing sociocentric networks. In low-dosage villages, treated individuals tend to sever health-advice ties with untreated alters, while in high-dosage villages treated individuals increase both inbound and outbound ties. The intervention also affects friendship and financial ties, indicating multiplex spillovers, yet aggregate village-level structure changes only modestly.}

\begin{rSection}{Manuscripts Under Review and Under Revision}

\pubtitle{Bayesian Predictive Synthesis for Dynamic Networks: Forecasting and Identifying Structural Mechanisms} \hfill {\em 2026}\\
\textit{(with Regina Ruane and Theofanis Papamichalis)}\\
{\small \textbf{Under revision, 2nd R\&R, \emph{Journal of the American Statistical Association}.} \myhref{https://arxiv.org/abs/2606.26136}{arXiv:2606.26136}.}\\
{\small \textbf{Abstract.} We combine competing generative models for dynamic networks through a predictive synthesis framework defined at the level of the graphon, so that forecasts are calibrated and the weights assigned to component models are interpretable as evidence for competing structural mechanisms of tie formation and dissolution.}

\pubtitle{Clustering Network Formation in Populations of Networks} \hfill {\em 2026}\\
\textit{(with Nicholas A.\ Christakis and Laura Forastiere)}\\
{\small \textbf{Under revision, 1st R\&R, \emph{Annals of Applied Statistics}.}}\\
{\small \textbf{Abstract.} We study inference for populations of networks with different node sets and sizes, where subsets of networks may share distinct formation regimes. Each observed network is modeled as a noisy snapshot of an underlying network, and networks are clustered jointly by a strategic formation mechanism and a measurement-error profile. We build a likelihood from a single snapshot using a meeting-and-revision equilibrium, with a finite-horizon alternative, and develop parallelizable Bayesian inference via block-updated MCMC with exchange-type updates.}

\pubtitle{Decision-Theoretic Robustness for Network Models} \hfill {\em 2026}\\
\textit{(with Regina Ruane, Swati Chandna, and Sim\'on Lunag\'omez)}\\
{\small \textbf{Under revision, \emph{Journal of Machine Learning Research}.}}\\
{\small \textbf{Abstract.} We develop a decision-theoretic notion of robustness for Bayesian network analysis by allowing the posterior to vary within a small Kullback--Leibler neighborhood. For exchangeable graphs, we connect robust posterior risk to graphon limits, yielding tractable sensitivity expansions for losses based on network functionals. We propose a practical computation via exponential tilting of posterior draws and illustrate the approach on brain-connectivity and village-network data.}

\begin{rSection}{Working Papers and Ongoing Projects}

\pubtitle{Peer Effects in Repeated Cooperation:
Evidence from 134 Honduran Villages} \hfill {\em 2026}\\
\textit{(with Theofanis Papamichalis, Nicholas A.\ Christakis and Feng Fu)}\\
{\small \textbf{Abstract.} Why does cooperation in real communities settle at moderate levels, neither converging to
full cooperation nor collapsing into free-riding? We study this question in a lab-in-the-field
experiment across 134 Honduran villages, where 2,591 participants play a ten-round publicgoods
game in anonymous, randomly assigned groups with an undisclosed horizon, and with a
complete pre-play friendship network. Using the experiment’s randomization, we find a moderate
but genuine peer effect: strong enough to draw every group toward a shared interior norm
near half the endowment, yet weak enough to leave them far from alike, so that 72 percent of
between-group differences survive to the final round. What sets cooperative groups apart is not
how they respond, but who is in them: the persistent gaps trace to the types each group brings
together, not to anything that happens during play. Cooperation here is something people bring,
not something the group creates.}

\pubtitle{Latent Space Network Modelling with Hyperbolic and Spherical Geometries} \hfill {\em 2026}\\
\textit{(with Kathryn Turnbull, Sim\'on Lunag\'omez, and Edoardo M.\ Airoldi)}\\
{\small \textbf{Abstract.} We propose latent space network models in hyperbolic and spherical geometries to capture non-Euclidean structure in relational data. We address identifiability by characterizing which latent representations are recoverable from observed links and by imposing constraints that resolve non-identifiability. We develop Bayesian estimation procedures, both MCMC and variational, tailored to each geometry, and evaluate the models on real networks.}
